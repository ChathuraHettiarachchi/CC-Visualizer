<objective>
## Goal
Two persistence/output features: (1) export — screenshot the 3D canvas as PNG and download session events as JSON; (2) bookmarks — bookmark any node with an optional note, stored in localStorage, visible in a new Bookmarks panel in the right rail.

## Purpose
Export lets developers share what they saw. Bookmarks let them annotate interesting moments in a session ("this is where it went wrong") and find them later across page refreshes.

## Output
- Export button on canvas (PNG screenshot via Three.js renderer, JSON download via blob URL)
- `lib/bookmarks.ts` — localStorage CRUD helpers keyed by sessionId+nodeId
- `components/panels/BookmarkPanel.tsx` — lists bookmarks; click flies camera to node
- RightRail adds "Bookmarks" panel below Files
- InspectorPanel gets a "Bookmark" toggle button when a node is selected
</objective>

    **lib/bookmarks.ts** (new file):
    ```ts
    export interface Bookmark {
      nodeId: string;
      nodeType: string | undefined;
      label: string;      // tool name or type
      note: string;
      timestamp: number;  // Date.now() when bookmarked
    }

    const key = (sessionId: string) => `cc-bookmarks-${sessionId}`;

    export function getBookmarks(sessionId: string): Bookmark[] {
      try {
        return JSON.parse(localStorage.getItem(key(sessionId)) ?? "[]");
      } catch { return []; }
    }

    export function addBookmark(sessionId: string, bookmark: Bookmark): void {
      const existing = getBookmarks(sessionId).filter(b => b.nodeId !== bookmark.nodeId);
      localStorage.setItem(key(sessionId), JSON.stringify([...existing, bookmark]));
    }

    export function removeBookmark(sessionId: string, nodeId: string): void {
      const existing = getBookmarks(sessionId).filter(b => b.nodeId !== nodeId);
      localStorage.setItem(key(sessionId), JSON.stringify(existing));
    }

    export function isBookmarked(sessionId: string, nodeId: string): boolean {
      return getBookmarks(sessionId).some(b => b.nodeId === nodeId);
    }
    ```
